\section{Setting and notation}
Pool of $n_0$ labelled samples; random hold-out $\mathcal{C}=\Ctr\cup\Cal$ with
$|\Cal|=m$; training set $D$ partitioned into folds $D_1,\dots,D_J$; fold model $f_{-j}$ trained
on $D\setminus D_j$; fused score $s_j(\cdot)$; null set $\Hnull$ (clean points). Code
correspondence: \texttt{cord.splits}, \texttt{cord.signals}, \texttt{cord.fusion},
\texttt{cord.pipeline}.

\begin{assumption}[A1, clean exchangeability]
Conditional on the poisoned samples, the clean points of each $D_j$ and the clean points of
$\Cal$ are exchangeable.
\end{assumption}

\begin{assumption}[A2, score invariance]
Within fold $j$, every score is a function of the scored point and of quantities that are
either independent of $(D_j,\Cal)$ or invariant to permutations of $D_j\cup\Cal$.
\end{assumption}

\section{Selection floor}
\begin{proposition}[Prop.~2]
The smallest conformal p-value is $1/(m+1)$ and every conformal e-value is at most $m+1$.
BH at level $\alpha$ and e-BH both select nothing unless the selected fraction satisfies
$\rho\ge\rho_{\mathrm{sel}}=1/(\alpha(m+1))$.
\end{proposition}
\begin{proof}
BH rejects $k$ of $N$ only if $p_{(k)}\le\alpha k/N$, hence only if $1/(m+1)\le\alpha k/N$,
i.e.\ $k/N\ge 1/(\alpha(m+1))$. e-BH selects $k$ of $n$ only if $e_{(k)}\ge n/(\alpha k)$, and
$e_{(k)}\le m+1$ gives the same inequality with $N=n$.

\end{proof}

\section{Presence test}
\begin{proposition}[Prop.~3]
Let $T_j^{(k)}$ be the number of points of $D_j$ whose score exceeds the $k$-th largest score in
$\Cal$. Under A1, A2 and no poison in $D_j\cup\Cal$, with $N=|D_j|$,
\[
\Pr\bigl(T_j^{(k)}\ge t\bigr)=F_{\mathrm{HG}}\bigl(k-1;\,N+m,\,m,\,t+k-1\bigr),
\qquad\text{in particular}\quad \Pr\bigl(T_j^{(1)}\ge t\bigr)=\binom{N}{t}\Big/\binom{N+m}{t},
\]
so $\pi_j=F_{\mathrm{HG}}(k-1;N+m,m,T_j+k-1)$ is a valid p-value and rejecting when
$\min_j\pi_j\le\alpha/J$ has false-alarm probability at most $\alpha$.
\end{proposition}
\begin{proof}[Proof structure]
Under exchangeability (and almost surely no ties) the ranks of the $N+m$ pooled scores are a
uniformly random permutation. $T^{(k)}\ge t$ iff at most $k-1$ calibration points are among the
top $t+k-1$ pooled ranks; the number of calibration points among those positions is
hypergeometric. Bonferroni over folds needs no assumption on their dependence.
\todo{Ties: strict exceedance makes the test conservative; write the argument. Implementation:
\texttt{cord.stats.presence}; exactness checked in \texttt{tests/test\_stats.py}.}
\end{proof}

\paragraph{Threshold.} Since $\binom{N}{t}/\binom{N+m}{t}\le(N/(N+m))^t$, the test rejects whenever
$T_j\ge \ln(J/\alpha)/\ln(1+m/N)$, i.e.\ at a rate of high-scoring poison
$\rho_{\mathrm{pres}}\approx\ln(J/\alpha)/m$ for $N\gg m$ (\texttt{cord floors}).

$\Pr(p_i\le t)\le t\,(1+m)/(1+m_{\mathrm{clean}})$; conjecture: $\mathrm{FDR}\le\alpha/(1-\varepsilon)$
up to $O(1/m)$.
